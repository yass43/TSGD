\section{Noise Assumptions}
\label{app:noise-assumptions}

\subsection{Bounded Variance and Similarity Are Incomparable}
\label{subsec:comparing-assumptions}

Write $h(x,\xi)=\nabla f(x)+r(x,\xi)$ with
$\mathbb E_\xi[r(x,\xi)]=0$. BV bounds
$\mathbb E_\xi[\|r(x,\xi)\|^2]$ uniformly in $x$, whereas global
similarity bounds
$\mathbb E_\xi[\|r(x,\xi)-r(y,\xi)\|^2]$ by
$\delta^2\|x-y\|^2$. The latter is sufficient for the conditional
trajectory condition in Assumption~\ref{ass:centered-similarity}
when each coupled difference uses a fresh sample. The conditional
condition may be weaker because it concerns only the generated
increments. The counterexamples below compare the two global oracle
conditions.

\paragraph{BV does not imply global similarity.}
Take $f(x)=x^2/2$ and $h(x,\xi)=x+\xi\sin(x^2)$, where $\xi$
takes the values $-1$ and $1$ with equal probability. The oracle is
unbiased and
\[
\mathbb E_\xi[|h(x,\xi)-\nabla f(x)|^2]
 =\sin^2(x^2)\leq1,
\]
so BV holds with $\sigma^2=1$. Under shared sampling,
\[
\mathbb E_\xi[|r(x,\xi)-r(y,\xi)|^2]
 =\bigl(\sin(x^2)-\sin(y^2)\bigr)^2.
\]
A finite similarity constant would imply
\[
\begin{aligned}
|\sin(x^2)-\sin(y^2)|&\leq\delta|x-y|
   &&\text{for every }x,y,\\
|2x\cos(x^2)|
 &=\lim_{y\to x}\frac{|\sin(y^2)-\sin(x^2)|}{|y-x|}
 \leq\delta &&\text{for every }x.
\end{aligned}
\]
But $x_n=\sqrt{2\pi n}$ gives
$|2x_n\cos(x_n^2)|=2\sqrt{2\pi n}\to\infty$, a contradiction.

\paragraph{Global similarity does not imply BV.}
Take $f(x)=\log\cosh x$ and
$h(x,\xi)=\tanh x+\xi x$, with
$\mathbb E[\xi]=0$ and $\mathbb E[\xi^2]=\delta^2>0$.
The objective is lower bounded and $1$-smooth, since
\[
f(x)\geq0,\qquad \nabla f(x)=\tanh x,\qquad
f''(x)=\frac1{\cosh^2x}\in(0,1].
\]
The unbiased oracle has residual $r(x,\xi)=\xi x$, and hence
\[
\mathbb E_\xi[|r(x,\xi)-r(y,\xi)|^2]
 =\delta^2(x-y)^2,
\qquad
\mathbb E_\xi[|r(x,\xi)|^2]=\delta^2x^2.
\]
Similarity holds with equality, but no finite global BV constant
exists. These examples establish incomparability of assumptions;
they do not by themselves order the performance of SGD and TSGD.
Appendix~\ref{app:least-squares} gives a natural instance: least squares
with a feature second-moment matrix of rank at least two satisfies
similarity but violates global BV.

\subsection{A TSGD Guarantee without Bounded Variance}
\label{subsec:similarity-without-bv}

For the log-cosh example above, $f_{\inf}=0$ and the fresh coupled
increment is
\[
D_{k+1}=\tanh x_{k+1}-\tanh x_k
       +\xi_{k+1}(x_{k+1}-x_k).
\]
Its conditional error variance is exactly
$\delta^2(x_{k+1}-x_k)^2$. Therefore, for deterministic $x_0$ and
arbitrary initialization with finite $E_0$,
\eqref{eq:tsgd-optimized-bound} gives
\begin{equation}
\label{eq:logcosh-tsgd-guarantee}
\overline G_K^{\mathrm{TSGD}}
\leq\frac{\log\cosh x_0}{K}
       \left(1+\sqrt{1+4(K-1)\delta^2}\right)+E_0.
\end{equation}
With exact initialization, this is an $O(K^{-1/2})$ stationarity
guarantee even though pointwise gradient variance is unbounded.

For comparison, fixed-step SGD satisfies
$x_{k+1}=x_k-\eta\tanh x_k-\eta\xi_kx_k$.
For $V_k:=\mathbb E[x_k^2]$, independence and centering give
\[
\begin{aligned}
V_{k+1}
 &=\mathbb E[(x_k-\eta\tanh x_k)^2]+\eta^2\delta^2V_k\\
 &\geq(1+\eta^2\delta^2)V_k-2\eta\sqrt{V_k},
\end{aligned}
\]
where $|\tanh x|\leq1$ and Cauchy--Schwarz were used. If
$|x_0|\geq4/(\eta\delta^2)$, then
\[
\sqrt{V_k}\geq\frac4{\eta\delta^2}
\quad\Longrightarrow\quad
2\eta\sqrt{V_k}\leq\frac{\eta^2\delta^2}{2}V_k
\quad\Longrightarrow\quad
V_{k+1}\geq\left(1+\frac{\eta^2\delta^2}{2}\right)V_k.
\]
Induction preserves the threshold and proves divergence of the
iterate second moment. This does not establish failure of stationarity,
which concerns $\mathbb E[\tanh^2x_k]$. Failure of BV alone is not
evidence that SGD cannot converge.

\subsection{Quadratic Finite Sums and the Smallest Similarity Constant}
\label{app:quadratic-similarity}

For $f_i(x)=\tfrac12x^\top A_ix+b_i^\top x+c_i$ with symmetric
$A_i$, let $f=n^{-1}\sum_i f_i$, $\bar A=n^{-1}\sum_i A_i$, and
sample $i$ uniformly. By \eqref{eq:quadratic-difference-noise},
the centered difference is $(A_i-\bar A)(x-y)$. For $z=x-y$,
\[
\mathbb E_i[\|(A_i-\bar A)z\|^2]
 =z^\top\!\left(\frac1n\sum_{i=1}^n(A_i-\bar A)^2\right)z.
\]
Taking the supremum over $\|z\|=1$ gives the smallest admissible
global constant,
\begin{equation}
\label{eq:quadratic-optimal-similarity}
\delta_{\min}^2
 =\lambda_{\max}\!\left(\frac1n\sum_{i=1}^n(A_i-\bar A)^2\right).
\end{equation}
The linear terms cancel irrespective of their magnitude, while the
smoothness constant of the average quadratic is
$L=\|\bar A\|_{\mathrm{op}}$. There is no fixed ordering between
$\delta$ and $L$. Identical $A_i$ give $\delta_{\min}=0$ even with
large $L$ and arbitrarily different $b_i$. Conversely, for $n\geq3$
scalar components, take $f_1(x)=nLx^2/2$ and $f_i(x)=0$ for $i\geq2$.
Then $f(x)=Lx^2/2$ but
\[
\delta_{\min}^2
 =\frac1n\bigl((nL-L)^2+(n-1)L^2\bigr)
 =(n-1)L^2>L^2.
\]
Thus similarity measures the discrepancy between stochastic and
deterministic gradient changes, rather than the smoothness of their
average.

\subsection{Least Squares}
\label{app:least-squares}

\paragraph{Setting.}
Let $(a,b)\in\mathbb R^d\times\mathbb R$ satisfy
$\mathbb E[\|a\|^4]<\infty$ and $\Sigma:=\mathbb E[aa^\top]\ne0$.
Assume $b=a^\top x^\star+\varepsilon$, where
\[
\mathbb E[\varepsilon\mid a]=0,\qquad
\mathbb E[\varepsilon^2\mid a]=s^2<\infty.
\]
Then the population loss and stochastic gradient satisfy
\[
\begin{aligned}
f(x)&=\tfrac12\mathbb E[(a^\top x-b)^2]
     =\tfrac12(x-x^\star)^\top\Sigma(x-x^\star)+\tfrac{s^2}{2},\\
h(x;a,b)&=a(a^\top x-b),\qquad
\nabla f(x)=\Sigma(x-x^\star).
\end{aligned}
\]
Thus Assumptions~\ref{ass:smooth-objective}--\ref{ass:unbiased-oracle}
hold with $L=\lambda_{\max}(\Sigma)>0$ and $f_{\inf}=s^2/2$.
At fixed $x_0,x^\star$ and feature distribution,
$\Delta_0=\tfrac12(x_0-x^\star)^\top\Sigma(x_0-x^\star)$ is independent of $s$.

\paragraph{Similarity and variance.}
The centered noise and its variation are
\begin{equation}
\label{eq:ls-noise}
\begin{aligned}
r(x;a,b)&=(aa^\top-\Sigma)(x-x^\star)-\varepsilon a,\\
r(x;a,b)-r(y;a,b)&=(aa^\top-\Sigma)(x-y).
\end{aligned}
\end{equation}
Define the positive semidefinite matrix
\[
M:=\mathbb E[(aa^\top-\Sigma)^2]
  =\mathbb E[\|a\|^2aa^\top]-\Sigma^2.
\]
Then $\mathbb E[\|r(x)-r(y)\|^2]=(x-y)^\top M(x-y)$,
which proves \eqref{eq:least-squares-similarity} by maximizing over
unit displacements. Conditional centering of $\varepsilon$ removes
the cross term in \eqref{eq:ls-noise}, giving
\begin{equation}
\label{eq:ls-variance}
\mathbb E[\|r(x)\|^2]
=(x-x^\star)^\top M(x-x^\star)+s^2\operatorname{tr}\Sigma.
\end{equation}
Label noise therefore adds $s^2\operatorname{tr}\Sigma$ at every point,
including minimizers, without changing $\delta$. If $\delta>0$, taking
$x-x^\star$ along a top eigenvector of $M$ makes this variance unbounded.
Conversely, $\delta=0$ implies $M=0$ and
$\mathbb E[\|aa^\top-\Sigma\|_{\mathrm F}^2]=\operatorname{tr}M=0$.
Hence $aa^\top=\Sigma$ almost surely, and BV holds with
$\sigma^2=s^2\operatorname{tr}\Sigma$. Since $\Sigma\ne0$ and $aa^\top$
has rank at most one, this requires $\operatorname{rank}\Sigma=1$.
Thus global BV holds exactly when $\delta=0$; rank one alone is not
sufficient. The variance is still bounded on every bounded set.

\paragraph{Feature variation.}
For $a\sim\mathcal N(0,\Sigma)$, Gaussian fourth moments give
\[
\mathbb E[\|a\|^2aa^\top]=\operatorname{tr}(\Sigma)\Sigma+2\Sigma^2,
\qquad
\delta^2=L\big(\operatorname{tr}\Sigma+L\big).
\]
Indeed, in an eigenbasis of $\Sigma$, the $i$th diagonal entry of
the expectation is
$3\lambda_i^2+\sum_{j\ne i}\lambda_i\lambda_j$; the off-diagonal
entries vanish. For $\Sigma=LI_d$, this yields $\delta^2=(d+1)L^2$.
With fixed feature norm $\|a\|^2=R^2$ and
$\Sigma=(R^2/d)I_d$, instead
$M=R^2\Sigma-\Sigma^2$ and $\delta^2=(d-1)L^2$.
Thus even normalized features can have $\delta/L$ of order $\sqrt d$.

\paragraph{Where label noise enters.}
Use independent samples across iterations and a fixed stepsize $\eta>0$.
Writing $z_k=x_k-x^\star$, the recursions are
\begin{equation}
\label{eq:ls-recursions}
\begin{aligned}
\mathrm{TSGD}:\quad
g_{k+1}&=(I-\eta a_{k+1}a_{k+1}^\top)g_k,
& z_{k+1}&=z_k-\eta g_k,\\
\mathrm{SGD}:\quad
z_{k+1}&=(I-\eta a_ka_k^\top)z_k+\eta\varepsilon_ka_k.
\end{aligned}
\end{equation}
The same random linear map acts on TSGD's estimator and SGD's iterate
error. Only SGD receives new label noise at every step. Given $x_0,g_0$
and subsequent features, TSGD needs no subsequent labels. In this population
model, exact initialization $g_0=\Sigma z_0$ also makes its trajectory
independent of $s$. For a batch of $B_0$ independent samples at $x_0$,
independent of all subsequent samples, \eqref{eq:ls-variance} gives
\begin{equation}
\label{eq:ls-initialization}
E_0=\frac{z_0^\top Mz_0+s^2\operatorname{tr}\Sigma}{B_0}.
\end{equation}
Fresh coupled differences satisfy
Assumptions~\ref{ass:unbiased-transport}--\ref{ass:centered-similarity}.
Consequently, Theorem~\ref{thm:tsgd-stationarity} applies for every finite
$s$; at fixed $x_0,x^\star$, feature distribution, and admissible stepsize,
its bound depends on label noise only through $E_0$.

\paragraph{Dependence on label-noise variance.}
At fixed $k,\eta,x_0,x^\star$, and feature distribution, the expected
squared gradient norms are affine in $s^2$:
\begin{equation}
\label{eq:ls-affine-errors}
G_k^{\mathrm{SGD}}=\alpha_k+s^2\beta_k,\qquad
G_k^{\mathrm{TSGD}}=\gamma_{k,B_0}+\frac{s^2}{B_0}\theta_k,
\end{equation}
with nonnegative coefficients independent of $s$. To see this for SGD,
let $S_k=\mathbb E[z_kz_k^\top]$ and
$\mathcal T(S)=\mathbb E[(I-\eta aa^\top)S(I-\eta aa^\top)]$. Then
\[
S_{k+1}=\mathcal T(S_k)+\eta^2s^2\Sigma,
\qquad G_k=\operatorname{tr}(\Sigma^2S_k).
\]
For TSGD, let $P_0=I$, $P_{t+1}=(I-\eta a_{t+1}a_{t+1}^\top)P_t$,
and $J_k=\eta\sum_{t<k}P_t$. Thus $z_k=z_0-J_kg_0$, with $J_k$
independent of $g_0$. Writing
$C_0=\mathbb E[(aa^\top-\Sigma)z_0z_0^\top(aa^\top-\Sigma)]$, we have
\[
\begin{aligned}
\mathbb E[g_0]&=\Sigma z_0,\qquad
\operatorname{Cov}(g_0)=\frac{C_0+s^2\Sigma}{B_0},\\
G_k^{\mathrm{TSGD}}
&=\mathbb E[\|\Sigma(z_0-J_k\Sigma z_0)\|^2]
 +\frac1{B_0}\mathbb E\!\left[
 \operatorname{tr}\big(\Sigma J_k(C_0+s^2\Sigma)J_k^\top\Sigma\big)\right].
\end{aligned}
\]
This proves \eqref{eq:ls-affine-errors}. Exact initialization removes
the covariance term entirely. The $1/B_0$ slope scaling holds at a
fixed iteration count; at a fixed evaluation budget, changing $B_0$
also changes the number of transport steps. Affinity alone does not
guarantee a crossover between the methods.

\paragraph{Finite sums.}
For a fixed dataset $\{(a_i,b_i)\}_{i=1}^n$, uniform sampling gives
$\Sigma=n^{-1}\sum_i a_ia_i^\top$. Label cancellation and the estimator
recursion in \eqref{eq:ls-recursions} remain exact, and
\eqref{eq:quadratic-optimal-similarity} gives $\delta$ with
$A_i=a_ia_i^\top$. However, the empirical minimizer $\widehat x^\star$
and exact initial gradient $g_0=\Sigma(x_0-\widehat x^\star)$ generally
depend on the labels. Thus exact initialization does not make the
trajectory invariant to changing a fixed dataset's labels; only the
subsequent transport is label independent given $x_0,g_0$.
