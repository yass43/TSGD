\section{Detailed Illustrative Examples}
\label{app:detailed-examples}

\subsection{Additive Noise: Exact Trajectories and Horizon Tuning}
\label{app:additive-details}

For explicit constants, specialize the additive example to
\begin{equation}
\label{eq:additive-comparison-oracle}
f(x)=\frac{x^2}{2},\qquad h(x,\xi)=x+\xi,\qquad
\mathbb P(\xi=1)=\mathbb P(\xi=-1)=\frac12,\qquad x_0=1.
\end{equation}
Here $L=1$, $\Delta_0=1/2$, $\sigma^2=1$, and $\delta=0$.
At $\eta=1$, SGD satisfies $x_{k+1}=-\xi_k$, hence
\begin{equation}
\label{eq:additive-sgd-fixed}
G_k^{\mathrm{SGD}}=1\quad(k\geq0),\qquad
\boxed{\overline G_K^{\mathrm{SGD}}=1.}
\end{equation}
Thus the fixed-step noise floor is present in the actual trajectory.

For the horizon-dependent BV stepsize
\eqref{eq:sgd-bv-optimal-step}, write
$\eta_K=2/(1+\sqrt{1+4K})$ and $r_K=1-\eta_K$.
Independence and centeredness of the fresh sample give
\[
G_{k+1}^{\mathrm{SGD}}
=\mathbb E[(r_Kx_k-\eta_K\xi_k)^2]
=r_K^2G_k^{\mathrm{SGD}}+\eta_K^2.
\]
Solving from $G_0=1$ and averaging the geometric terms yields
\begin{align*}
G_k^{\mathrm{SGD}}
&=r_K^{2k}+\frac{\eta_K}{2-\eta_K}(1-r_K^{2k}),\\
\overline G_K^{\mathrm{SGD}}
&=\frac{\eta_K}{2-\eta_K}
+\left(1-\frac{\eta_K}{2-\eta_K}\right)
\frac{1-r_K^{2K}}{K(1-r_K^2)}.
\end{align*}
Since $\eta_K\sim K^{-1/2}$,
$r_K^{2K}\leq e^{-2K\eta_K}\to0$ and
$1-r_K^2=\eta_K(2-\eta_K)\sim2K^{-1/2}$.
Each term in the average is therefore asymptotic to $1/(2\sqrt K)$:
\begin{equation}
\label{eq:additive-sgd-tuned}
\boxed{\overline G_K^{\mathrm{SGD}}=\frac{1+o(1)}{\sqrt K}.}
\end{equation}
The optimized BV bound is $(1+\sqrt{1+4K})/(2K)$, so both its
rate and leading constant are attained by this prescribed run.

For TSGD, use $\eta=1$ and any square-integrable $g_0=1+e_0$.
Exact differences give $D_{k+1}=x_{k+1}-x_k=-g_k$;
therefore $g_1=0$ and $x_k=-e_0$ for all $k\geq1$. Hence
\begin{equation}
\label{eq:additive-tsgd-exact}
\boxed{\overline G_K^{\mathrm{TSGD}}
=\frac1K+\left(1-\frac1K\right)E_0.}
\end{equation}
With exact initialization, TSGD reaches the minimizer after one
step. For fixed $E_0>0$, horizon-tuned SGD eventually has the smaller
error. The comparison depends on initialization as well as transport.

\subsection{Multiplicative Noise: Fresh Gradients and False Convergence}
\label{app:multiplicative-details}

For \eqref{eq:multiplicative-comparison-oracle},
$\mathbb E[a]=1$ and $\mathbb E[(a-1)^2]=1$. The oracle is
unbiased and its pointwise noise variance is $x^2$, so global BV
does not hold. Coupled centered differences nevertheless satisfy
\[
\mathbb E[|(a-1)(x-y)|^2]=|x-y|^2,
\]
giving $\delta=1$. These comparisons use exact dynamics rather
than the BV theorem.

At $\eta=1/2$, SGD obeys $x_{k+1}=(1-a_k/2)x_k$;
it remains at one until the first sample $a=2$, then stays at zero.
Thus
\begin{equation}
\label{eq:multiplicative-sgd-exact}
\boxed{G_k^{\mathrm{SGD}}=2^{-k},\qquad
\overline G_K^{\mathrm{SGD}}=\frac{2(1-2^{-K})}{K}.}
\end{equation}
For TSGD at the same step, let $g_0=\nabla f(x_0)=1$ and
$N:=\min\{n\geq1:a_n=2\}$. The estimator recursion is
$g_{k+1}=(1-a_{k+1}/2)g_k$, so
\[
\mathbb P(N=n)=2^{-n},\qquad
g_k=\boldsymbol 1_{\{k<N\}},\qquad
x_k=1-\frac12\min\{k,N\}.
\]
Since $N<\infty$ almost surely and $\mathbb E[N^2]=6$,
dominated convergence gives $x_k\to1-N/2$ in mean square.
Using $\mathbb E[N]=2$,
\begin{equation}
\label{eq:multiplicative-tsgd-residual}
\boxed{\lim_{k\to\infty}G_k^{\mathrm{TSGD}}
=\mathbb E[(1-N/2)^2]=\frac12.}
\end{equation}
The estimator becomes zero almost surely in finite time, but the
limiting true gradient is nonzero with probability
$1-\mathbb P(N=2)=3/4$. Ces\`aro averaging also gives
$\overline G_K^{\mathrm{TSGD}}\to1/2$.

\subsection{Sharpness for Horizon-Tuned TSGD}
\label{app:proof-multiplicative-tsgd-tuned}

For the same oracle with exact initialization,
\eqref{eq:tsgd-optimal-step} gives
$\eta_K^\star=2/(1+\sqrt{4K-3})$, since
$L=\delta=1$ and $\Delta_0=1/2$.
The optimized guarantee becomes
\begin{equation}
\label{eq:multiplicative-tsgd-tuned-bound}
\overline G_K^{\mathrm{TSGD}}
\leq\frac{1+\sqrt{4K-3}}{2K}.
\end{equation}

\begin{lemma}[Asymptotic sharpness for tuned TSGD]
\label{lem:multiplicative-tsgd-tuned-exact}
For \eqref{eq:multiplicative-comparison-oracle}, with exact
initialization and the constant step $\eta_K^\star$ for the
$K$-iteration run,
\[
\lim_{K\to\infty}\sqrt K\,\overline G_K^{\mathrm{TSGD}}=1.
\]
\end{lemma}

\begin{proof}
Fix $K\geq2$ and write $\eta=\eta_K^\star$. The updates imply
\[
g_{k+1}=(1-\eta a_{k+1})g_k,\qquad
e_{k+1}=e_k+\eta(1-a_{k+1})g_k.
\]
Set $C_k:=\mathbb E[g_ke_k]$ and
$q:=\mathbb E[(1-\eta a)^2]=1-2\eta+2\eta^2$.
Independence yields
\[
M_{k+1}=qM_k,\qquad E_{k+1}=E_k+\eta^2M_k,
\qquad C_{k+1}=(1-\eta)C_k+\eta^2M_k.
\]
Starting from $M_0=1$ and $E_0=C_0=0$, these give
\[
M_k=q^k,\qquad E_k=\frac{\eta}{2(1-\eta)}(1-q^k),\qquad
C_k=\eta^2\sum_{t=0}^{k-1}(1-\eta)^{k-1-t}q^t.
\]
Since $0<\eta<1$ and $q<1$, all summands are nonnegative and
\[
\sum_{k<K}C_k
\leq\eta^2\left(\sum_{t\geq0}q^t\right)
             \left(\sum_{j\geq0}(1-\eta)^j\right)
=\frac{\eta^2}{(1-q)\eta}
=\frac1{2(1-\eta)}.
\]
Thus the average cross moment is $O(K^{-1})$ as
$\eta_K^\star\to0$. For the other moments,
\[
\frac1K\sum_{k<K}M_k=\frac{1-q^K}{K(1-q)},\qquad
\frac1K\sum_{k<K}E_k=
\frac{\eta}{2(1-\eta)}
\left(1-\frac1K\sum_{k<K}M_k\right).
\]
Now $\eta_K^\star\sim K^{-1/2}$,
$1-q=2\eta_K^\star(1-\eta_K^\star)\sim2K^{-1/2}$, and
$q^K\leq e^{-K(1-q)}\to0$. Therefore
\[
\frac1K\sum_{k<K}M_k\sim\frac1{2\sqrt K},\qquad
\frac1K\sum_{k<K}E_k\sim\frac1{2\sqrt K}.
\]
Finally, $x_k=g_k-e_k$ implies
$G_k=M_k+E_k-2C_k$. The averaged cross term is lower order,
which proves the claim.
\end{proof}

The theorem's bound has the correct leading rate and constant for
this prescribed TSGD run. Fixed-step SGD in
\eqref{eq:multiplicative-sgd-exact} has average error
$\Theta(K^{-1})$, whereas this horizon-tuned TSGD run has
$\Theta(K^{-1/2})$ error. This is not a lower bound over all TSGD
stepsize schedules.

\subsection{Fixed Variance and Variable Similarity}
\label{app:sinusoidal-proof}

\begin{lemma}[Fixed variance and variable similarity]
\label{lem:fixed-variance-variable-similarity}
For \eqref{eq:fixed-variance-oracle}, the noise is centered, satisfies
\eqref{eq:phase-noise-moments}, and has smallest global similarity
constant $\delta=\sigma\omega$ for $\omega\geq0$.
\end{lemma}

\begin{proof}
A uniform phase makes the cosine have mean zero and second moment
$1/2$, so
$\mathbb E[r(x,\varphi)^2]=2\sigma^2(1/2)=\sigma^2$.
For the coupled difference,
\[
r(x,\varphi)-r(y,\varphi)
=-2\sqrt2\sigma
\sin\!\left(\frac{\omega(x+y)}2+\varphi\right)
\sin\!\left(\frac{\omega(x-y)}2\right).
\]
Squaring and using
$\mathbb E_\varphi[\sin^2(c+\varphi)]=1/2$ for every fixed $c$ gives
\[
\mathbb E[|r(x,\varphi)-r(y,\varphi)|^2]
=8\sigma^2\cdot\frac12\,
  \sin^2\!\left(\frac{\omega(x-y)}2\right)
=4\sigma^2\sin^2\!\left(\frac{\omega(x-y)}2\right).
\]
This verifies \eqref{eq:phase-noise-moments}.
The inequality $|\sin t|\leq|t|$ proves similarity with
$\delta=\sigma\omega$; minimality follows from
\[
\lim_{y\to x}
\frac{4\sigma^2\sin^2(\omega(x-y)/2)}{(x-y)^2}
=\sigma^2\omega^2.
\]
\end{proof}

SGD's fresh noise has conditional variance $\sigma^2$ at every
iterate, giving the frequency-independent exact recursion
\begin{equation}
\label{eq:phase-sgd-dynamics}
G_{k+1}^{\mathrm{SGD}}
=(1-L\eta)^2G_k^{\mathrm{SGD}}+L^2\eta^2\sigma^2.
\end{equation}
Indeed, even its conditional noise distribution is independent of
$x_k$ and $\omega$, since adding $\omega x_k$ to a uniform phase
preserves its distribution modulo $2\pi$. For TSGD,
\begin{equation}
\label{eq:phase-tsgd-error}
\begin{aligned}
E_{k+1}-E_k
&=4\sigma^2\mathbb E\!\left[
\sin^2\!\left(\frac{\omega(x_{k+1}-x_k)}2\right)\right]\\
&\leq\sigma^2\omega^2\eta^2M_k.
\end{aligned}
\end{equation}
For fixed $\sigma>0$, varying $\omega$ realizes any smallest global
similarity constant without changing the pointwise noise distribution.
The sine factor also shows that
increasing $\omega$ need not increase every realized increment's
error monotonically.

\subsection{Numerical Illustration of Fixed Variance and Variable Similarity}
\label{app:sinusoid}

The preceding construction lets us vary transport accuracy without
changing the distribution of individual stochastic gradients.
Figure~\ref{fig:sinusoid} illustrates the resulting difference between
SGD and TSGD. We fix $L=\sigma=x_0=1$ and $\eta=0.1$, and vary
$\omega\in\{0,0.5,1,2\}$, so $\delta=\omega$.
SGD has the same law at every frequency. TSGD follows GD at $\omega=0$,
but accumulates transport error at nonzero frequencies. In these runs,
its final error is below SGD's at $\omega=0.5$ and above it at
$\omega=2$, despite identical pointwise noise distributions.

\begin{figure}[t]
\centering
\includegraphics[width=0.9\linewidth]{figures/similarity_experiment.pdf}
\caption{Same pointwise variance, different transport accuracy.
Monte Carlo means over $10^5$ runs, with bands of two standard errors;
$L=\sigma=x_0=1$ and a common constant stepsize $\eta=0.1$.
Every method initializes by averaging gradients at phases $\varphi_0$
and $\varphi_0+\pi$: their noises cancel, giving $g_0=Lx_0$.
These two evaluations are included in the horizontal-axis count.
Subsequent SGD updates cost one evaluation and TSGD updates cost two.
SGD is simulated at $\omega=1$; its expected curve is identical at
all frequencies.}
\label{fig:sinusoid}
\end{figure}

\paragraph{Sampling and initialization.}
\label{app:numerical-details}

Each curve averages $100{,}000$ independent trajectories. To isolate
transport error, all methods start from the same exact estimate,
obtained using the oracle's antithetic symmetry:
\[
g_0=\frac12\bigl(h(x_0,\varphi_0)+h(x_0,\varphi_0+\pi)\bigr)
=Lx_0.
\]
This exact cancellation relies on paired phases, not independent
samples. Subsequent phases are independent and uniform, with the
same phase used at both endpoints of each transport difference.

\paragraph{Evaluation count.}
The horizontal axis includes the two evaluations used to initialize
each method. All methods use this estimate for their first update. Thereafter,
SGD uses one new stochastic gradient per update and TSGD uses two
pointwise evaluations per coupled difference. After $k\geq1$
iterate updates, the costs are $k+1$ for SGD and $2k$ for TSGD.
No unused final transport correction is computed. Hence a budget
of $200$ permits $199$ SGD updates and $100$ TSGD updates.
The comparison therefore starts from identical information, but relies
on an initialization procedure specific to this oracle.

\paragraph{Constant-step behavior.}
For the sinusoidal quadratic, conditional centeredness and
similarity give
\[
\mathbb E[g_{k+1}^2\mid\mathcal F_k]
\leq\bigl((1-\eta)^2+\eta^2\omega^2\bigr)g_k^2
\leq0.85g_k^2.
\]
Thus the estimator contracts in mean square at every tested frequency.
The experiment illustrates constant-step error accumulation, not
the horizon-optimized guarantee of Section~\ref{sec:tsgd-convergence}.
The SGD recurrence \eqref{eq:phase-sgd-dynamics} applies after the
common first update, for which $G_1=0.9^2$. Every simulated SGD
mean is within $3.35$ estimated standard errors of this exact
recurrence. At $\omega=0$, TSGD follows
$G_k=0.9^{2k}$; the relative numerical discrepancy is below
$10^{-12}$ throughout the plotted run.

\paragraph{Results.}
At budget $200$, the final sample means and standard errors are
\begin{center}
\small
\begin{tabular}{@{}lrr@{}}
\toprule
Method & Mean squared true gradient & Standard error\\
\midrule
SGD & $0.052791$ & $0.000226$\\
TSGD, $\omega=0$ & $7.0551\times10^{-10}$ & below $10^{-20}$\\
TSGD, $\omega=0.5$ & $0.013409$ & $0.000063$\\
TSGD, $\omega=1$ & $0.055485$ & $0.000331$\\

\end{center}
The means include all trajectories, including rare large errors:
at $\omega=2$, the largest $0.1\%$ of final squared gradients contribute
$7.72\%$ of the total. Shaded bands show pointwise Monte Carlo uncertainty
in the mean, not the spread of individual trajectories. The observed
ordering illustrates the importance of noise variation; it does not
assert that every transport increment becomes noisier as $\omega$
increases, since its variance depends on the sine term in
\eqref{eq:phase-noise-moments}.


\subsection{Least-Squares Experiments}
\label{app:ls}

The least-squares experiments illustrate four aspects of transport:
label-noise cancellation, dependence on initialization, dependence on
similarity, and false convergence. We use one model throughout, varying
label noise and feature variability separately. These are comparisons
at a common stepsize and iteration count, not comparisons of computational
cost or optimally tuned performance.

\paragraph{Model.}
Let $f(x)=\tfrac12\mathbb E[\|Ax-b\|^2]$ in $d=10$ dimensions, with
$H=\operatorname{diag}(0.1,0.2,\ldots,1)$,
$v=1+\kappa\zeta$ with $\mathbb P(\zeta=1)=\mathbb P(\zeta=-1)=1/2$,
$A=\sqrt v\,H^{1/2}$, and $b=Ax^\star+s\varepsilon$, where
$x^\star=0$ and $\varepsilon\sim\mathcal N(0,I_d)$ is independent
of $v$. For $0\leq\kappa<1$, the stochastic gradient and its shared-sample
difference are
\[
\begin{aligned}
h(x;v,\varepsilon)&=vH(x-x^\star)-s\sqrt v\,H^{1/2}\varepsilon,\\
h(x;v,\varepsilon)-h(y;v,\varepsilon)&=vH(x-y).
\end{aligned}
\]
Hence $\nabla f(x)=H(x-x^\star)$ and
\[
\begin{aligned}
\mathbb E[\|h(x)-\nabla f(x)\|^2]
  &=\kappa^2\|H(x-x^\star)\|^2+s^2\operatorname{tr}(H),\\
\mathbb E[\|r(x)-r(y)\|^2]&=\kappa^2\|H(x-y)\|^2.
\end{aligned}
\]
Thus $s$ controls the additive contribution to gradient noise but does
not affect transport. The smallest global similarity constant is
$\delta=\kappa$, since $L=\|H\|=1$.

\paragraph{Protocol.}
The initial point is
$x_0=\boldsymbol 1/\|H\boldsymbol 1\|$, so
$\|\nabla f(x_0)\|^2=1$. Each run uses $\eta=0.5$ and 400 updates;
the figures show the first 80 iterations. Curves average 4,000 independent
trajectories, with pointwise bands of $\pm2$ Monte Carlo standard errors.
Initialization samples are independent of subsequent samples. Each
transport increment uses the same feature and label at both endpoints.

\paragraph{Sampled initialization: preserving a good or bad estimate.}
Figure~\ref{fig:ls-init} uses $s=1$ and $\kappa=0.05$: fresh gradients
are noisy, whereas their differences are accurate. SGD uses one fresh
sample per update. TSGD starts from an average of
$B_0\in\{1,128,512\}$ independent stochastic gradients, giving
\[
E_0=\frac{\kappa^2\|Hx_0\|^2+s^2\operatorname{tr}(H)}{B_0},
\]
The observed residuals are approximately $5.51$, $0.044$, and $0.011$,
respectively. A single-sample initialization gives a larger residual than
SGD, while the more accurate initial estimates give smaller residuals.
Accurate transport therefore benefits from good initial information,
but does not repair a poor starting estimate. Increasing $B_0$ also costs
more initial evaluations; this figure illustrates its statistical effect,
not a computational advantage.

\paragraph{Noiseless labels: false convergence.}
Figure~\ref{fig:ls-main}(b) instead uses $s=0$ and $\kappa=0.9$,
with $B_0=128$ for TSGD. Fresh-gradient noise now vanishes at the
minimizer, and SGD's expected squared gradient norm tends to zero.
The TSGD estimator also tends to zero in mean square, but its expected
squared true-gradient norm approaches approximately $0.268$.
Indeed, the coordinatewise second moments of SGD's true gradient and
TSGD's estimator share the contraction factors
$\rho_j=(1-\eta\lambda_j)^2+\eta^2\kappa^2\lambda_j^2$.
Their curves nearly coincide because TSGD's sampled initialization
increases each initial second moment by only a factor
$1+\kappa^2/B_0\approx1.0063$. The agreement is between SGD's true
gradient and TSGD's estimator, not between their stationarity errors.

\paragraph{Controlled initialization: isolating noise cancellation and similarity.}
Figure~\ref{fig:ls-main}(a) starts both methods from the exact gradient,
with $\kappa=0.2$ and matched feature draws across $s\in\{0,1,5\}$.
TSGD's trajectories coincide across all three label-noise levels because
the shared label noise cancels from every increment. Its positive
residual is due to feature variability, not label noise. By contrast,
SGD's residual increases with $s$ and vanishes when $s=0$.
Figure~\ref{fig:ls-errors} separately varies $\delta$ and a prescribed
initial error $E_0$; Appendix~\ref{app:quad-numerics} derives its exact
limiting errors. Unlike the sampled initializations above, these
controlled estimates isolate mechanisms without modeling how the initial
accuracy is obtained.
