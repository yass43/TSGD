\section{Detailed Convergence Proofs}
\label{app:convergence-proofs}

\subsection{Descent and Weighted Accumulation}
\label{subsec:descent-and-accumulation}

Fix deterministic $x_0$, a horizon $K\geq1$, and a constant step
$\eta>0$. In addition to the notation of
Section~\ref{sec:tsgd-convergence}, write
$q_k:=\mathbb E[\|d_{k+1}\|^2]$.
The finite second moment of $g_0$, smoothness, and conditional
similarity imply by induction that $x_k,g_k,e_k$ have finite second
moments at every finite $k$. Smoothness also bounds the growth of
$f$ by a quadratic, so the expectations below are well defined.
The assumption $L>0$ excludes only a trivial case: a lower-bounded
function with $0$-Lipschitz gradient is constant.

For completeness, $e_k$ is $\mathcal F_k$-measurable and
$\mathbb E[d_{k+1}\mid\mathcal F_k]=0$. Therefore
\[
\mathbb E[\langle e_k,d_{k+1}\rangle]
 =\mathbb E[\langle e_k,
                 \mathbb E[d_{k+1}\mid\mathcal F_k]\rangle]=0,
\qquad E_{k+1}=E_k+q_k.
\]
Iteration yields \eqref{eq:transport-error-identity}; more generally,
distinct transport errors are orthogonal in expectation. The process
$(e_k)$ is a square-integrable martingale for the history containing
the initialization and all transport samples observed by step $k$.

The smoothness inequality and the identity
\[
2\langle\nabla f(x_k),g_k\rangle
 =\|\nabla f(x_k)\|^2+\|g_k\|^2-\|e_k\|^2
\]
give
\[
\begin{aligned}
f(x_{k+1})
&\leq f(x_k)-\eta\langle\nabla f(x_k),g_k\rangle
                  +\frac{L\eta^2}{2}\|g_k\|^2\\
&=f(x_k)-\frac\eta2\|\nabla f(x_k)\|^2
        -\frac\eta2(1-L\eta)\|g_k\|^2
        +\frac\eta2\|e_k\|^2.
\end{aligned}
\]
Taking expectations proves \eqref{eq:tsgd-fundamental-descent}.
The martingale identity \eqref{eq:transport-error-identity} yields
\begin{equation}
\label{eq:weighted-error-accumulation}
\sum_{k=0}^{K-1}E_k
 =KE_0+\sum_{k=0}^{K-1}\sum_{t=0}^{k-1}q_t
 =KE_0+\sum_{t=0}^{K-2}(K-1-t)q_t.
\end{equation}
The last sum is empty for $K=1$. Similarity gives
\begin{equation}
\label{eq:transport-error-step-bound}
q_t\leq\delta^2\mathbb E[\|x_{t+1}-x_t\|^2]
     =\delta^2\eta^2M_t.
\end{equation}
The weight $K-1-t$ counts the later estimates that inherit the error
generated by the $t$th displacement.

\subsection{Proof of Theorem~\ref{thm:tsgd-stationarity}}

Summing \eqref{eq:tsgd-fundamental-descent} and substituting
\eqref{eq:weighted-error-accumulation} gives
\[
F_K+\frac\eta2\sum_{k=0}^{K-1}G_k
 +\frac\eta2(1-L\eta)\sum_{k=0}^{K-1}M_k
\leq\Delta_0+\frac{\eta K}{2}E_0
 +\frac\eta2\sum_{t=0}^{K-2}(K-1-t)q_t.
\]
Apply \eqref{eq:transport-error-step-bound} and collect the coefficients
of $M_k$. For
$c_k:=1-L\eta-\delta^2\eta^2(K-1-k)$, this proves the stronger bound
\begin{equation}
\label{eq:tsgd-full-bound}
\frac1K\sum_{k=0}^{K-1}G_k
 +\frac{2F_K}{\eta K}
 +\frac1K\sum_{k=0}^{K-1}c_kM_k
\leq\frac{2\Delta_0}{\eta K}+E_0.
\end{equation}
Condition \eqref{eq:tsgd-step-condition} ensures $c_k\geq0$ for every
$0\leq k<K$. Since also $F_K\geq0$, dropping the last two terms on
the left proves \eqref{eq:tsgd-stationarity-bound}.

\subsection{Exact Optimized TSGD Guarantee}
\label{subsec:optimized-tsgd-guarantees}

The largest positive step satisfying
$L\eta+(K-1)\delta^2\eta^2\leq1$ is
\begin{equation}
\label{eq:tsgd-optimal-step}
\boxed{\eta^\star
 =\frac{2}{L+\sqrt{L^2+4(K-1)\delta^2}}.}
\end{equation}
For $(K-1)\delta^2>0$, this is the positive root of the quadratic,
written after rationalizing its numerator. The same expression gives
$1/L$ when $K=1$ or $\delta=0$. It maximizes the step under the
theorem's sufficient condition; it is not a necessary stability limit.
Substitution into \eqref{eq:tsgd-stationarity-bound} yields
\begin{equation}
\label{eq:tsgd-optimized-bound}
\overline G_K^{\mathrm{TSGD}}
\leq\frac{\Delta_0}{K}
   \left(L+\sqrt{L^2+4(K-1)\delta^2}\right)+E_0.
\end{equation}
Using
\[
\sqrt{L^2+4(K-1)\delta^2}\leq L+2\delta\sqrt{K-1},
\qquad \frac{\sqrt{K-1}}K\leq\frac1{\sqrt K},
\]
gives \eqref{eq:tsgd-simplified-bound}.
For $(K-1)\delta^2\ll L^2$, $\eta^\star$ is close to $1/L$;
for $(K-1)\delta^2\gg L^2$, it is of order $1/(\delta\sqrt K)$.
These are the deterministic and transport-dominated regimes of this
bound. Exact initialization removes $E_0$ but does not remove the
transport term; exact transport removes the latter but preserves $E_0$.

\subsection{SGD under Bounded Variance}
\label{app:sgd-convergence}

Assume \ref{ass:smooth-objective}--\ref{ass:bounded-variance}, and
let $F_k,G_k$ now refer to the SGD iterates. Conditional unbiasedness
and BV imply
\[
\mathbb E_\xi[\|h(x_k,\xi)\|^2\mid x_k]
\leq\|\nabla f(x_k)\|^2+\sigma^2.
\]
Consequently,
\[
F_{k+1}\leq F_k-\eta\left(1-\frac{L\eta}{2}\right)G_k
                    +\frac{L\eta^2\sigma^2}{2}.
\]
Summing and dropping $F_K\geq0$ yields, for $0<\eta<2/L$,
\begin{equation}
\label{eq:sgd-bv-bound}
\boxed{\overline G_K^{\mathrm{SGD}}
\leq\frac{2\Delta_0+KL\eta^2\sigma^2}{\eta K(2-L\eta)}.}
\end{equation}
In particular, the fixed gradient-descent step $\eta=1/L$ gives
\begin{equation}
\label{eq:sgd-bv-fixed-step}
\overline G_K^{\mathrm{SGD}}
\leq\frac{2L\Delta_0}{K}+\sigma^2.
\end{equation}

For $\Delta_0>0$, let
$U(\eta):=(2\Delta_0/K+L\sigma^2\eta^2)/(\eta(2-L\eta))$.
Its derivative has the sign of
\[
L\sigma^2\eta^2+\frac{2L\Delta_0}{K}\eta
                   -\frac{2\Delta_0}{K}.
\]
This expression is strictly increasing for $\eta>0$ and changes sign
once in $(0,2/L)$. Thus the minimizing constant step is
\begin{equation}
\label{eq:sgd-bv-optimal-step}
\boxed{\eta_{\mathrm{BV},K}^*
 =\frac{2}{L+\sqrt{L^2+2KL\sigma^2/\Delta_0}}.}
\end{equation}
At the root,
$L\sigma^2\eta^2=(2\Delta_0/K)(1-L\eta)$, so
$U(\eta)=2\Delta_0/(K\eta)$. Hence
\begin{equation}
\label{eq:sgd-bv-optimized-bound}
\boxed{\overline G_K^{\mathrm{SGD}}
\leq\frac{\Delta_0}{K}
 \left(L+\sqrt{L^2+\frac{2KL\sigma^2}{\Delta_0}}\right)
\leq\frac{2L\Delta_0}{K}
     +\sqrt{\frac{2L\Delta_0\sigma^2}{K}}.}
\end{equation}
The last inequality uses $\sqrt{a+b}\leq\sqrt a+\sqrt b$.
When $\sigma^2=0$, the step is $1/L$ and the bound reduces to
$2L\Delta_0/K$. When $\sigma^2>0$, the step decreases with $K$,
giving a finite-horizon guarantee rather than an assertion about an
infinite run with a fixed positive step. If $\Delta_0=0$, $x_0$ is
already stationary; the optimized formula above is not used.

\subsection{Iteration Complexity}
\label{subsec:tsgd-complexity}

Let $R$ be uniform on $\{0,\ldots,K-1\}$, independent of the run.
Then $\mathbb E[\|\nabla f(x_R)\|^2]=\overline G_K$.

\begin{corollary}[Sufficient TSGD horizon]
\label{cor:tsgd-iteration-complexity}
Under Theorem~\ref{thm:tsgd-stationarity}, suppose
$E_0\leq\varepsilon^2/3$ and use $\eta=\eta^\star$. An integer
horizon satisfying
\begin{equation}
\label{eq:tsgd-iteration-complexity}
K\geq\max\left\{1,\frac{6L\Delta_0}{\varepsilon^2},
                      \frac{36\delta^2\Delta_0^2}{\varepsilon^4}\right\}
\end{equation}
ensures $\mathbb E[\|\nabla f(x_R)\|^2]\leq\varepsilon^2$.
\end{corollary}
\begin{proof}
Each of the first two terms in \eqref{eq:tsgd-simplified-bound} is
at most $\varepsilon^2/3$ under \eqref{eq:tsgd-iteration-complexity};
the initialization term satisfies the same bound by assumption.
\end{proof}

For SGD with $\Delta_0>0$ and step \eqref{eq:sgd-bv-optimal-step},
the analogous sufficient horizon is
\begin{equation}
\label{eq:sgd-iteration-complexity}
K\geq\max\left\{1,\frac{4L\Delta_0}{\varepsilon^2},
                      \frac{8L\Delta_0\sigma^2}{\varepsilon^4}\right\}.
\end{equation}
Each of the two terms in the simplified SGD bound is then at most
$\varepsilon^2/2$. These are iteration upper bounds; initialization
and per-iteration oracle costs are accounted for in
Appendix~\ref{app:oracle-complexity}.
