\section{Exact Quadratic Dynamics and Error Sources}
\label{app:quadratic-dynamics}
\label{sec:two-error-sources}

\subsection{Model and Recursions}
\label{subsec:quadratic-model}

Let $x_0$ be deterministic and
\[
f(x)=\frac{\lambda x^2}{2},\qquad
h(x;a,b)=(\lambda+a)x+b,\qquad \lambda>0,
\]
where $a,b$ are independent, centered, and
$\mathbb E[a^2]=\delta^2$, $\mathbb E[b^2]=\sigma^2<\infty$.
Use independent fresh pairs at successive steps, independently of
the square-integrable $g_0$. Shared sampling cancels $b$, giving
$D_{k+1}=(\lambda+a_{k+1})(x_{k+1}-x_k)$ and
\begin{equation}
\label{eq:quadratic-core-recursions}
g_{k+1}=(1-\eta\lambda-\eta a_{k+1})g_k,\qquad
e_{k+1}=e_k-\eta a_{k+1}g_k.
\end{equation}
Thus conditional similarity holds with equality, independently of
$\sigma$. Write $M_k=\mathbb E[|g_k|^2]$,
$E_k=\mathbb E[|e_k|^2]$, and $G_k=\mathbb E[|\lambda x_k|^2]$.

\subsection{Exact Transport}
\label{subsec:exact-init-exact-transport}
\label{subsec:inexact-init-exact-transport}

For any objective, exact differences preserve $e_k=e_0$. TSGD is
therefore GD on $\widetilde f(x)=f(x)+\langle e_0,x\rangle$,
and is ordinary GD when $e_0=0$.
At $\eta=1/L$ and $e_0=0$, descent gives, for every $K$,
\[
\frac1{2L}\sum_{k<K}\|\nabla f(x_k)\|^2\leq\Delta_0.
\]
This proves the usual $2L\Delta_0/K$ averaged guarantee and, by
boundedness of the nonnegative partial sums,
$\sum_{k\geq0}\|\nabla f(x_k)\|^2<\infty$.
Hence each gradient tends to zero, without convexity or convergence
of the iterates. For fixed nonzero $e_0$, the same argument on
$\widetilde f$ gives $\nabla f(x_k)\to-e_0$ if $\widetilde f$ is
also bounded below. The finite-horizon bound
$2L\Delta_0/K+E_0$ requires no such additional lower bound.

For the quadratic, put $\delta=0$ and $r=1-\eta\lambda$.
Equation~\eqref{eq:quadratic-core-recursions} and
$\lambda x_k=g_k-e_0$ give
\begin{equation}
\label{eq:quadratic-tilted-trajectory}
g_k=r^k(\lambda x_0+e_0),\qquad
x_k=r^kx_0-\frac{1-r^k}{\lambda}e_0.
\end{equation}
For centered $e_0$,
$G_k=r^{2k}\lambda^2x_0^2+(1-r^k)^2E_0$.
For any square-integrable $e_0$, biased or not, $|r|<1$ gives
$x_k\to-e_0/\lambda$ in mean square and $G_k\to E_0$.

\paragraph{Sharpness of the initialization coefficient.}
Choose $E_0>0$ and $\eta=1/\lambda$.
Then $G_k=E_0$ for every $k\geq1$, so
\[
\frac1K\sum_{k<K}G_k
=\frac{\lambda^2x_0^2}{K}+\left(1-\frac1K\right)E_0
\longrightarrow E_0.
\]
The bound $2\Delta_0/(\eta K)+E_0$ has the same limit. Thus the
coefficient one on $E_0$ is sharp.

\subsection{Stochastic Transport and the Interaction Formula}
\label{subsec:exact-init-stochastic-transport}
\label{subsec:inexact-init-stochastic-transport}

For arbitrary square-integrable initialization, set
$\rho=(1-\eta\lambda)^2+\eta^2\delta^2$.
Independence and centering in \eqref{eq:quadratic-core-recursions} give
\begin{equation}
\label{eq:quadratic-moment-recursions}
M_{k+1}=\rho M_k,\qquad E_{k+1}=E_k+\eta^2\delta^2M_k.
\end{equation}
Consequently,
\[
M_k=\rho^kM_0,\qquad
E_k=E_0+\eta^2\delta^2M_0\sum_{t<k}\rho^t.
\]
Assume the mean-square stability condition
\begin{equation}
\label{eq:quadratic-stability}
0<\eta<\frac{2\lambda}{\lambda^2+\delta^2},
\qquad\text{equivalently }\rho<1.
\end{equation}
With $A:=\eta^2\delta^2/(1-\rho)$,
$A(1-\rho)=\eta^2\delta^2$ and \eqref{eq:quadratic-moment-recursions}
imply the conservation law and its limit:
\begin{equation}
\label{eq:quadratic-conservation}
E_k+AM_k=E_0+AM_0,\qquad
E_\infty:=\lim_kE_k=E_0+AM_0.
\end{equation}

To identify the limiting true-gradient error, use the orthogonality
of the martingale increments. For $m>n$,
\[
\mathbb E[|e_m-e_n|^2]
=\eta^2\delta^2\sum_{t=n}^{m-1}M_t
\leq\frac{\eta^2\delta^2M_0}{1-\rho}\rho^n\to0.
\]
Thus $(e_k)$ is Cauchy in the complete space $L^2$ and converges in
mean square to $e_\infty$. Since $g_k\to0$ in mean square,
$\lambda x_k=g_k-e_k\to-e_\infty$ in mean square, proving
\begin{equation}
\label{eq:quadratic-interaction}
\boxed{\lim_kG_k=E_\infty
=E_0+\frac{\eta\delta^2}{2\lambda-\eta(\lambda^2+\delta^2)}M_0.}
\end{equation}
In particular, exact initialization gives $E_0=0$ and
$M_0=\lambda^2x_0^2$. If $\delta>0$ and $x_0\ne0$, the estimator
vanishes in mean square while the true gradient has a positive
limiting mean-square norm.

Initialization contributes directly through $E_0$ and indirectly through
$M_0=\lambda^2x_0^2+2\lambda x_0\mathbb E[e_0]+E_0$.
Bias may increase or decrease the latter. For centered $e_0$,
\begin{equation}
\label{eq:quadratic-three-contributions}
E_\infty
=\underbrace{E_0}_{\text{initialization}}
+\underbrace{A\lambda^2x_0^2}_{\text{transport}}
+\underbrace{AE_0}_{\text{interaction}}.
\end{equation}
This yields the four cases in Table~\ref{tab:four-error-regimes}.
The stochastic-transport formulas use the quadratic model; the
exact-transport identities hold for general objectives.
