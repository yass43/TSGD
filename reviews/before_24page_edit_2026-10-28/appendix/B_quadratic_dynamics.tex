\section{Exact Quadratic Dynamics and Error Sources}
\label{app:quadratic-dynamics}
\label{sec:two-error-sources}

\subsection{The Quadratic Model}
\label{subsec:quadratic-model}

Let $x_0$ be deterministic, $\eta>0$ constant, and
\begin{equation}
\label{eq:quadratic-noise-model}
f(x)=\frac\lambda2x^2,\qquad
h(x;a,b)=(\lambda+a)x+b,\qquad \lambda>0.
\end{equation}
The noises are independent and satisfy
\[
\mathbb E[a]=\mathbb E[b]=0,\qquad
\mathbb E[a^2]=\delta^2,\qquad \mathbb E[b^2]=\sigma^2.
\]
Thus $\nabla f(x)=\lambda x$ and $\mathbb E[h(x;a,b)]=\nabla f(x)$.
Each transport step uses a fresh pair $(a_{k+1},b_{k+1})$,
independent of the past and the square-integrable initial estimate
$g_0$, at both endpoints. The additive noise cancels:
\begin{equation}
\label{eq:quadratic-coupled-difference}
D_{k+1}=(\lambda+a_{k+1})(x_{k+1}-x_k).
\end{equation}
Consequently,
\[
\mathbb E\bigl[|D_{k+1}-[\nabla f(x_{k+1})-\nabla f(x_k)]|^2
       \mid\mathcal F_k\bigr]
=\delta^2|x_{k+1}-x_k|^2.
\]
Assumption~\ref{ass:centered-similarity} holds with equality at
$\delta=\sqrt{\mathbb E[a^2]}$, independently of $\sigma$.
Substituting $x_{k+1}-x_k=-\eta g_k$ in $g_{k+1}=g_k+D_{k+1}$
gives
\begin{equation}
\label{eq:quadratic-core-recursions}
g_{k+1}=(1-\eta\lambda-\eta a_{k+1})g_k,
\qquad e_{k+1}=e_k-\eta a_{k+1}g_k.
\end{equation}
We retain $E_k=\mathbb E[|e_k|^2]$, $M_k=\mathbb E[|g_k|^2]$,
and $G_k=\mathbb E[|\nabla f(x_k)|^2]$. In particular, tracking
error $E_k$ and true stationarity error $G_k$ are distinct quantities.

\subsection{Exact Initialization and Exact Transport}
\label{subsec:exact-init-exact-transport}

Exact initialization and transport give $g_k=\nabla f(x_k)$ by
induction, so TSGD is GD path by path. At $\eta=1/L$, GD descent yields
\[
\frac1{2L}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2
\leq f(x_0)-f(x_K)\leq\Delta_0.
\]
Consequently,
\begin{equation}
\label{eq:exact-transport-gd-bound}
\frac1K\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2\leq\frac{2L\Delta_0}{K}.
\end{equation}
The partial sums are uniformly bounded, so the infinite series
converges and $\|\nabla f(x_k)\|^2\to0$, without convexity or
convergence of the iterates.

For the quadratic, $\delta=0$ means $a=0$ almost surely.
Writing $r=1-\eta\lambda$, exact initialization gives
\begin{equation}
\label{eq:quadratic-exact-gd}
x_k=r^kx_0,\qquad g_k=\lambda r^kx_0,\qquad E_k=0.
\end{equation}
Thus $G_k=\lambda^2r^{2k}x_0^2\to0$ for $0<\eta<2/\lambda$.

\subsection{Inexact Initialization and Exact Transport}
\label{subsec:inexact-init-exact-transport}

For any objective, let $g_0=\nabla f(x_0)+e_0$ and take exact
differences. Then $e_k=e_0$ for every $k$, and
\[
g_k=\nabla f(x_k)+e_0=\nabla\widetilde f_{e_0}(x_k),
\qquad \widetilde f_{e_0}(x):=f(x)+\langle e_0,x\rangle.
\]
Hence TSGD is GD on a fixed linearly tilted objective, and $E_k=E_0$.
At $\eta=1/L$, Theorem~\ref{thm:tsgd-stationarity} gives
\begin{equation}
\label{eq:exact-transport-inexact-init-bound}
\frac1K\sum_{k=0}^{K-1}G_k\leq\frac{2L\Delta_0}{K}+E_0.
\end{equation}
For fixed $e_0$, if $\widetilde f_{e_0}$ is also bounded below,
GD descent on that objective implies
$\nabla f(x_k)\to-e_0$ and
$\|\nabla f(x_k)\|^2\to\|e_0\|^2$.
This additional lower bound is required for the pathwise asymptotic
claim, not for \eqref{eq:exact-transport-inexact-init-bound}.

For the quadratic and $\delta=0$, the exact dynamics are
\begin{equation}
\label{eq:quadratic-tilted-trajectory}
g_k=r^k(\lambda x_0+e_0),\qquad
x_k=r^kx_0-\frac{1-r^k}{\lambda}e_0.
\end{equation}
The first identity follows from \eqref{eq:quadratic-core-recursions};
the second follows from $\lambda x_k=g_k-e_0$.
If $\mathbb E[e_0]=0$, then
\begin{equation}
\label{eq:quadratic-centered-init-stationarity}
G_k=r^{2k}\lambda^2x_0^2+(1-r^k)^2E_0.
\end{equation}
For $0<\eta<2/\lambda$, even with biased square-integrable $e_0$,
$x_k\to-e_0/\lambda$ in mean square and
\begin{equation}
\label{eq:quadratic-init-residual}
\lim_{k\to\infty}G_k=E_0.
\end{equation}
\paragraph{Sharpness of the initialization coefficient.}
Choose an initialization with $E_0>0$.
For $\varepsilon>0$, choose $N$ with $|G_k-E_0|\leq\varepsilon$
for $k\geq N$. For every $K>N$,
\[
\left|\frac1K\sum_{k=0}^{K-1}G_k-E_0\right|
\leq\frac1K\sum_{k=0}^{N-1}|G_k-E_0|+\varepsilon.
\]
Letting $K\to\infty$ and then $\varepsilon\downarrow0$ gives,
at the admissible $\eta=1/L=1/\lambda$,
\[
\lim_{K\to\infty}\frac1K\sum_{k=0}^{K-1}G_k
=E_0
=\lim_{K\to\infty}\left(\frac{2\Delta_0}{\eta K}+E_0\right).
\]
Thus the coefficient one on $E_0$ in
\eqref{eq:tsgd-stationarity-bound} is sharp.

\subsection{Exact Initialization and Stochastic Transport}
\label{subsec:exact-init-stochastic-transport}

Take $g_0=\lambda x_0$ and $\delta>0$, and define
\[
r:=1-\eta\lambda,\qquad
\rho:=\mathbb E[(r-\eta a)^2]
=r^2+\eta^2\delta^2
=1-2\eta\lambda+\eta^2(\lambda^2+\delta^2).
\]
Independence and centeredness of $a_{k+1}$ give
\begin{equation}
\label{eq:quadratic-moment-recursions}
M_{k+1}=\rho M_k,\qquad E_{k+1}=E_k+\eta^2\delta^2M_k.
\end{equation}
Starting from $E_0=0$ and $M_0=\lambda^2x_0^2$,
\begin{equation}
\label{eq:quadratic-exact-init-moments}
M_k=\rho^k\lambda^2x_0^2,\qquad
E_k=\eta^2\delta^2\lambda^2x_0^2\sum_{t=0}^{k-1}\rho^t.
\end{equation}
The stability condition is
\begin{equation}
\label{eq:quadratic-stability}
\rho<1\quad\Longleftrightarrow\quad
0<\eta<\frac{2\lambda}{\lambda^2+\delta^2}.
\end{equation}
Under this condition, $g_k\to0$ in mean square, while
\begin{equation}
\label{eq:quadratic-exact-init-residual}
E_\infty:=\lim_{k\to\infty}E_k
=\frac{\eta\delta^2\lambda^2x_0^2}
       {2\lambda-\eta(\lambda^2+\delta^2)}.
\end{equation}
To relate this tracking error to the true gradient, note that
$e_{t+1}-e_t=-\eta a_{t+1}g_t$ are orthogonal martingale increments.
For $m>n$,
\[
\mathbb E[|e_m-e_n|^2]
=\eta^2\delta^2\sum_{t=n}^{m-1}M_t
\leq\frac{\eta^2\delta^2M_0}{1-\rho}\rho^n\longrightarrow0.
\]
Thus $(e_k)$ is Cauchy in the complete space $L^2$, and converges
in mean square to some $e_\infty$. Since $g_k\to0$ in mean square,
$x_k=(g_k-e_k)/\lambda\to-e_\infty/\lambda=:x_\infty$ in mean
square as well. Consequently,
\[
\lim_{k\to\infty}M_k=0,\qquad
\lim_{k\to\infty}G_k
=\mathbb E[|\nabla f(x_\infty)|^2]
=\mathbb E[|e_\infty|^2]=E_\infty>0\quad(x_0\ne0).
\]
The estimator converges to zero in mean square while the true
gradient does not. This is the false-convergence phenomenon.

\subsection{Inexact Initialization and Stochastic Transport}
\label{subsec:inexact-init-stochastic-transport}

Allow arbitrary square-integrable $e_0$ and retain
\eqref{eq:quadratic-stability}. The moment recursions remain valid.
Define
\begin{equation}
\label{eq:quadratic-A}
A:=\frac{\eta^2\delta^2}{1-\rho}
=\frac{\eta\delta^2}{2\lambda-\eta(\lambda^2+\delta^2)}.
\end{equation}
Since $A(1-\rho)=\eta^2\delta^2$,
\[
E_{k+1}+AM_{k+1}
=E_k+(\eta^2\delta^2+A\rho)M_k=E_k+AM_k.
\]
Induction proves the conservation law
\begin{equation}
\label{eq:quadratic-conservation}
E_k+AM_k=E_0+AM_0\qquad(k\geq0).
\end{equation}
Since $M_k=\rho^kM_0\to0$, the exact interaction formula follows:
\begin{equation}
\label{eq:quadratic-interaction}
\boxed{E_\infty=E_0+
\frac{\eta\delta^2}{2\lambda-\eta(\lambda^2+\delta^2)}M_0
=E_0+AM_0.}
\end{equation}
The preceding $L^2$ argument also gives $G_k\to E_\infty$.
Initialization contributes directly through $E_0$ and indirectly
through the initial estimator magnitude
\[
M_0=\lambda^2x_0^2+2\lambda x_0\mathbb E[e_0]+E_0.
\]
Bias can therefore increase or decrease the transport contribution.
For $g_0=\lambda x_0+b_0$, with centered $b_0$ independent of future
samples and $\mathbb E[b_0^2]=\sigma^2$, the three contributions are
\begin{equation}
\label{eq:quadratic-three-contributions}
E_\infty
=\underbrace{\sigma^2}_{\text{initialization}}
+\underbrace{A\lambda^2x_0^2}_{\text{transport}}
+\underbrace{A\sigma^2}_{\text{interaction}}.
\end{equation}
In Table~\ref{tab:four-error-regimes}, the four corresponding limiting
stationarity errors are respectively $0$, $A\lambda^2x_0^2$,
$E_0$, and $E_0+AM_0$. The exact-transport identities are general;
these stochastic-transport formulas rely on the quadratic model.

\subsection{Initialization and Transport Errors on Least Squares}
\label{app:quad-numerics}

Figure~\ref{fig:ls-errors} separates the effects of initialization and
similarity in the least-squares model of Appendix~\ref{app:ls}.
We prescribe $g_0=Hx_0+e_0$, with
$e_0\sim\mathcal N(0,(E_0/d)I_d)$ independent of subsequent samples.
Each coordinate follows the scalar quadratic recursion with curvature
$\lambda_j$ and noise variance $\kappa^2\lambda_j^2$, so summing
\eqref{eq:quadratic-interaction} gives
\begin{equation}
\label{eq:least-squares-residual}
\lim_{k\to\infty}\mathbb E[\|\nabla f(x_k)\|^2]
=E_0+\sum_{j=1}^d
\frac{\eta\kappa^2\lambda_j}{2-\eta(1+\kappa^2)\lambda_j}
\left(\lambda_j^2x_{0,j}^2+\frac{E_0}{d}\right),
\end{equation}
provided $0<\eta<2/[(1+\kappa^2)\max_j\lambda_j]$, as in our runs.

We compare $\delta=\kappa\in\{0.05,0.9\}$ and
$E_0\in\{10^{-3},10^{-1}\}$.
The observed plateaus agree with the exact limits shown by dotted lines.
Better initialization reduces both the inherited error and its
contribution to later transport error; smaller $\delta$ reduces the
transport contribution. Neither improvement replaces the other:
accurate initialization alone does not prevent a large transport
residual. These controlled initializations isolate the two effects
without modeling the cost of obtaining $g_0$.
