\section{Noise Assumptions}
\label{app:noise-assumptions}

\subsection{Bounded Variance, ABC, and Centered Similarity}
\label{subsec:comparing-assumptions}

Write $h(x,\xi)=\nabla f(x)+r(x,\xi)$ with
$\mathbb E_\xi[r(x,\xi)]=0$.

\paragraph{ABC generalizes BV.}
BV requires the gradient-noise variance to be uniformly bounded over
the domain. The more general ABC condition, introduced as expected
smoothness by \citet{khaled2023better}, assumes that some
$A,B,C\geq0$ satisfy
\begin{equation}
\label{eq:abc-assumption}
\mathbb E_\xi[\|h(x,\xi)\|^2]
\leq2A(f(x)-f_{\inf})+B\|\nabla f(x)\|^2+C
\qquad\text{for all }x.
\end{equation}
By unbiasedness, BV gives $(A,B,C)=(0,1,\sigma^2)$.
ABC allows the second moment to grow with the objective gap and
gradient norm. If $C=0$, and both $f(x_k)-f_{\inf}\to0$ and
$\|\nabla f(x_k)\|^2\to0$, then
$\mathbb E_\xi[\|h(x_k,\xi)\|^2]\to0$: the second-moment penalty
in SGD's descent bound can vanish.

For example, let $f(x)=x^2/2$ and $h(x,a)=ax$, where $a$ equals
$0$ or $2$ with equal probability. Its residual is
$r(x,a)=h(x,a)-\nabla f(x)=(a-1)x$, so
\[
\mathbb E[h(x,a)]=x,\qquad
\mathbb E[|h(x,a)|^2]=2x^2,\qquad
\mathbb E[|r(x,a)|^2]=x^2.
\]
Thus ABC holds with $(A,B,C)=(0,2,0)$, while global BV fails
because the noise variance $x^2$ is unbounded as $|x|\to\infty$,
even though fresh gradients are exact at the minimizer.

\paragraph{One-point and two-point conditions.}
BV and ABC constrain the stochastic gradient at each point separately.
Centered similarity constrains noise differences under shared sampling.
Section~\ref{subsec:fixed-variance-variable-similarity} shows that
identical pointwise distributions can have different similarity
constants. The following global counterexamples establish more:
neither BV nor ABC implies similarity, and similarity implies neither.

\paragraph{Neither BV nor ABC implies global similarity.}
Take $f(x)=x^2/2$ and $h(x,\xi)=x+\xi\sin(x^2)$, where
$\mathbb P(\xi=-1)=\mathbb P(\xi=1)=1/2$. The oracle is
unbiased and
\[
\mathbb E_\xi[|h(x,\xi)-\nabla f(x)|^2]
 =\sin^2(x^2)\leq1,
\]
so BV holds with $\sigma^2=1$, and ABC holds with $(A,B,C)=(0,1,1)$.
Under shared sampling,
\[
\mathbb E_\xi[|r(x,\xi)-r(y,\xi)|^2]
 =\bigl(\sin(x^2)-\sin(y^2)\bigr)^2.
\]
A finite similarity constant would imply
\[
\begin{aligned}
|\sin(x^2)-\sin(y^2)|&\leq\delta|x-y|
   &&\text{for every }x,y,\\
|2x\cos(x^2)|
 &=\lim_{y\to x}\frac{|\sin(y^2)-\sin(x^2)|}{|y-x|}
 \leq\delta &&\text{for every }x.
\end{aligned}
\]
However, at $x_n=\sqrt{2\pi n}$, $n\geq1$, this would require
\[
|2x_n\cos(x_n^2)|=2\sqrt{2\pi n}\leq\delta
\qquad\text{for every }n.
\]
For any finite $\delta$, choosing an integer $n>\delta^2/(8\pi)$
violates this inequality. Hence no finite global similarity constant
exists.

\paragraph{Global similarity implies neither BV nor ABC.}
Take $f(x)=\log\cosh x$ and
$h(x,\xi)=\tanh x+\xi x$, with
$\mathbb E[\xi]=0$ and $\mathbb E[\xi^2]=\delta^2>0$.
The objective is nonnegative and $1$-smooth, since
$f''(x)=1/\cosh^2x\in(0,1]$.
The unbiased oracle has residual $r(x,\xi)=\xi x$, and hence
\[
\mathbb E_\xi[|r(x,\xi)-r(y,\xi)|^2]
 =\delta^2(x-y)^2,
\qquad
\mathbb E_\xi[|r(x,\xi)|^2]=\delta^2x^2.
\]
Similarity holds with equality, but no finite global BV constant
exists. Since $f_{\inf}=0$, ABC would require
\[
\tanh^2x+\delta^2x^2
\leq2A\log\cosh x+B\tanh^2x+C
\leq2A|x|+B+C
\qquad\text{for all }x.
\]
For $x\neq0$, dividing by $x^2$ and dropping the nonnegative term
$\tanh^2x/x^2$ gives
\[
\delta^2\leq\frac{2A}{|x|}+\frac{B+C}{x^2}.
\]
For any finite nonnegative $A,B,C$, the right-hand side tends to zero as
$|x|\to\infty$, whereas the left-hand side is the fixed positive
constant $\delta^2$. The inequality therefore fails for sufficiently
large $|x|$, so no finite ABC constants exist.

For this oracle, Theorem~\ref{thm:tsgd-stationarity} still applies:
the coupled increment has conditional error variance
$\delta^2(x_{k+1}-x_k)^2$. Thus exact initialization gives the
$O(K^{-1/2})$ TSGD guarantee despite failure of both BV and ABC.
This does not by itself establish that SGD fails.

\subsection{Quadratic Finite Sums and the Smallest Similarity Constant}
\label{app:quadratic-similarity}

For $f_i(x)=\tfrac12x^\top A_ix+b_i^\top x+c_i$ with symmetric
$A_i$, let $f=\frac1n\sum_{i=1}^n f_i$,
$\bar A=\frac1n\sum_{i=1}^n A_i$, and sample $i$ uniformly.
The centered gradient difference is
\begin{equation}
\label{eq:quadratic-difference-noise}
[\nabla f_i(x)-\nabla f_i(y)]-[\nabla f(x)-\nabla f(y)]
=(A_i-\bar A)(x-y).
\end{equation}
For $z=x-y$,
\[
\mathbb E_i[\|(A_i-\bar A)z\|^2]
 =z^\top\!\left(\frac1n\sum_{i=1}^n(A_i-\bar A)^2\right)z.
\]
Taking the supremum over $\|z\|=1$ gives the smallest admissible
global constant,
\begin{equation}
\label{eq:quadratic-optimal-similarity}
\delta_{\min}^2
 =\lambda_{\max}\!\left(\frac1n\sum_{i=1}^n(A_i-\bar A)^2\right).
\end{equation}
The linear terms cancel irrespective of their magnitude, while the
smoothness constant of the average quadratic is
$L=\|\bar A\|_{\mathrm{op}}$. There is no fixed ordering between
$\delta$ and $L$. Identical $A_i$ give $\delta_{\min}=0$ even with
large $L$ and arbitrarily different $b_i$. Conversely, for $n\geq3$
scalar components, take $f_1(x)=nLx^2/2$ and $f_i(x)=0$ for $i\geq2$.
Then $f(x)=Lx^2/2$ but
\[
\delta_{\min}^2
 =\frac1n\bigl((nL-L)^2+(n-1)L^2\bigr)
 =(n-1)L^2>L^2.
\]
Thus similarity measures the discrepancy between stochastic and
deterministic gradient changes, rather than the smoothness of their
average.
